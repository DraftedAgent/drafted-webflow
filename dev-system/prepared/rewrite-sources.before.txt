// NODE Normalize output
// Normalize model response -> expects { updates: [...] }
// Run once for each item

const src = $json;

function first(x) { return Array.isArray(x) ? x[0] : x; }

const root = first(src.output ?? src);

let rawText =
  root?.content?.[0]?.text ??
  root?.content?.[0]?.output_text ??
  root?.text ??
  "";

// clean
rawText = String(rawText)
  .replace(/^\uFEFF/, "")
  .replace(/^=+/, "")
  .trim();

if (!rawText) {
  return { json: { ok: false, error: "Empty response from model" } };
}

let parsed;
try {
  parsed = JSON.parse(rawText);
} catch (e) {
  return { json: { ok: false, error: "Model output is not valid JSON", raw: rawText } };
}

if (!Array.isArray(parsed.updates)) {
  return { json: { ok: false, error: "Missing updates[] in model output", parsed } };
}

const updates = parsed.updates
  .map(u => ({
    blockId: String(u?.blockId ?? "").trim(),
    replacementText: String(u?.replacementText ?? "")
  }))
  .filter(u => u.blockId && u.replacementText.length > 0);

if (!updates.length) {
  // Det är OK att modellen inte vill ändra något, men Webflow vill fortfarande få blocks[] tillbaka.
  return { json: { ok: true, updates: [] } };
}

return { json: { ok: true, updates } };

// NODE Validation + Normalization
// Code node - Run once for all items
// BLOCK-ONLY: mode = "blocks" | "full"
// Expects: instruction, blocks[], (optional) selectedBlockIds[], targetRole, cvTitle, cvVersionId

const items = $input.all();

function bad(msg, extra = {}) {
  return [{ json: { ok: false, error: String(msg), ...extra } }];
}

const first = items[0]?.json ?? {};
const body = first.body ?? first.data ?? first;

const modeRaw = String(body.mode ?? "").trim().toLowerCase();
const mode = (modeRaw === "blocks" || modeRaw === "full") ? modeRaw : "blocks";

// Required
const instruction = String(body.instruction ?? "").trim();
if (!instruction) return bad("Instruction is empty");

// Common optional fields (pass-through)
const targetRole  = String(body.targetRole ?? "").trim();
const cvTitle     = String(body.cvTitle ?? "").trim();
const cvVersionId = String(body.cvVersionId ?? "").trim() || null;

// Blocks required
const blocksIn = Array.isArray(body.blocks) ? body.blocks : null;
if (!blocksIn || blocksIn.length === 0) return bad("Missing field: blocks[]");

// Normalize blocks minimally + validate blockId
const blocks = blocksIn.map((b, idx) => ({
  blockId: String(b?.blockId ?? "").trim(),
  type: String(b?.type ?? "").trim(),
  label: String(b?.label ?? "").trim(),
  order: Number.isFinite(Number(b?.order)) ? Number(b.order) : idx * 10,
  text: String(b?.text ?? ""),
  employer: b?.employer ?? undefined,
  title: b?.title ?? undefined,
  startDate: b?.startDate ?? undefined,
  endDate: b?.endDate ?? undefined,
  institution: b?.institution ?? undefined,
  degree: b?.degree ?? undefined,
  program: b?.program ?? undefined
}));

const missingId = blocks.find(b => !b.blockId);
if (missingId) return bad("One or more blocks missing blockId");

// Selection for blocks-mode
let selectedBlockIds = Array.isArray(body.selectedBlockIds)
  ? body.selectedBlockIds.map(x => String(x).trim()).filter(Boolean)
  : [];

if (mode === "blocks") {
  if (!selectedBlockIds.length) return bad("Missing field: selectedBlockIds[] for mode=blocks");

  const selectedSet = new Set(selectedBlockIds);

  // Ensure all selected exist
  const existing = new Set(blocks.map(b => b.blockId));
  const unknown = selectedBlockIds.filter(id => !existing.has(id));
  if (unknown.length) {
    return bad("selectedBlockIds contains unknown blockId(s)", { unknown });
  }

  // Filter down to selected only (cheaper for model)
  const selectedBlocks = blocks.filter(b => selectedSet.has(b.blockId));
  if (!selectedBlocks.length) return bad("No blocks matched selectedBlockIds");

  return [{
    json: {
      ok: true,
      mode: "blocks",
      instruction,
      targetRole,
      cvTitle,
      cvVersionId,
      selectedBlockIds: Array.from(selectedSet),
      blocks: selectedBlocks,
      // IMPORTANT: also keep originalBlocks so we can merge updates later in n8n
      originalBlocks: blocks
    }
  }];
}

// full-mode: rewrite whole CV (still blocks)
return [{
  json: {
    ok: true,
    mode: "full",
    instruction,
    targetRole,
    cvTitle,
    cvVersionId,
    selectedBlockIds: [],
    blocks,           // send all
    originalBlocks: blocks
  }
}];

// NODE Apply updates + Return main-format
// Apply updates onto originalBlocks (FULL LIST) and return main-format response

const base = $node["Validation + Normalization"].json;

// FULL base list must be originalBlocks
const originalBlocks = Array.isArray(base.originalBlocks) ? base.originalBlocks : [];

// updates from Normalize output
const updates = Array.isArray($json.updates) ? $json.updates : [];

// build map: blockId -> replacementText
const byId = new Map(
  updates
    .filter(u => u && typeof u === "object")
    .map(u => [String(u.blockId || "").trim(), String(u.replacementText ?? "")])
    .filter(([id]) => !!id)
);

// apply (keep every block)
const nextBlocks = originalBlocks.map(b => {
  const id = String(b?.blockId || "").trim();
  if (!id) return b;

  if (!byId.has(id)) return b;

  return {
    ...b,
    text: byId.get(id)
  };
});

// Build rewrittenCv from blocks (single source of truth)
// IMPORTANT: do NOT mutate nextBlocks order (sort mutates), so copy first.
function joinBlocksToText(blocks) {
  const list = Array.isArray(blocks) ? blocks.slice() : [];
  list.sort((a, b) => (a?.order ?? 0) - (b?.order ?? 0));
  return list
    .map(b => String(b?.text ?? ""))
    .join("\n\n")
    .trim();
}

const rewrittenCv = joinBlocksToText(nextBlocks);

return {
  json: {
    ok: true,
    blocksSchemaVersion: "1.0",
    cvVersionId: base.cvVersionId || null,
    cvTitle: base.cvTitle || "",
    summaryOfChanges: "",
    blocks: nextBlocks,
    rewrittenCv,
    rewrittenCvLegacy: rewrittenCv
  }
};
